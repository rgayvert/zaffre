const resolveOptionMapEntry = (map: OptionMap, key: string) => {
  if (typeof map[key] === "function") {
    // use a regex to capture the argument inside brackets
    const result = key.match(/\w+\[(\w+)\]/);
    if (result && result[0] === key) {
      const arg = result[1];
      return map[key](arg);
    }
  }
  return <Attributes>map[key];
};
